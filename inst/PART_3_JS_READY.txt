    let audioContext = null;


    function systemSound(type){

      try{

        const AudioCtx =
          window.AudioContext ||
          window.webkitAudioContext;

        if(!AudioCtx) return;


        if(!audioContext){
          audioContext = new AudioCtx();
        }


        if(audioContext.state === "suspended"){
          audioContext.resume();
        }


        const oscillator =
          audioContext.createOscillator();

        const gain =
          audioContext.createGain();

        const now =
          audioContext.currentTime;


        let frequency = 520;


        if(type === "close"){
          frequency = 330;
        }


        if(type === "secret"){
          frequency = 720;
        }


        oscillator.type = "sine";

        oscillator.frequency.setValueAtTime(
          frequency,
          now
        );


        if(type === "secret"){

          oscillator.frequency
            .exponentialRampToValueAtTime(
              980,
              now + .09
            );

        }


        gain.gain.setValueAtTime(
          .0001,
          now
        );

        gain.gain.exponentialRampToValueAtTime(
          .012,
          now + .008
        );

        gain.gain.exponentialRampToValueAtTime(
          .0001,
          now + .11
        );


        oscillator.connect(gain);

        gain.connect(
          audioContext.destination
        );


        oscillator.start(now);

        oscillator.stop(
          now + .13
        );


      }catch(error){}

    }

    function updateNotificationStack(){

    }

    function enhanceNotification(item){

      if(
        !item ||
        item.dataset.dismissReady === "1"
      ){
        return;
      }


      item.dataset.dismissReady =
        "1";


      const close =
        document.createElement("button");

        "button";

      close.className =
        "ih-notification-close";

      close.setAttribute(
        "aria-label",
        "Close notification"
      );

        "×";


      close.addEventListener(
        "click",
        function(event){

          event.preventDefault();
          event.stopPropagation();

          dismissNotification(
            item,
            "right"
          );

        }
      );


      item.appendChild(close);


      let startX = 0;
      let currentX = 0;
      let dragging = false;


      item.addEventListener(
        "pointerdown",
        function(event){

          if(
            event.target.closest(
              ".ih-notification-close"
            )
          ){
            return;
          }


          startX =
            event.clientX;

          currentX =
            0;

          dragging =
            true;


          item.classList.add(
            "ih-swiping"
          );


          try{

            item.setPointerCapture(
              event.pointerId
            );

          }catch(error){}

        }
      );


      item.addEventListener(
        "pointermove",
        function(event){

          if(!dragging) return;


          currentX =
            event.clientX -
            startX;


          const resistance =
            Math.abs(currentX) > 140
            ?
            .72
            :
            1;


          const x =
            currentX *
            resistance;


          item.style.transform =
            "translateX(" +
            x +
            "px)";


          item.style.opacity =
            Math.max(
              .35,
              1 -
              Math.abs(x) /
              230
            );

        }
      );


      function finishSwipe(){

        if(!dragging) return;


        dragging =
          false;


        const threshold =
          Math.min(
            90,
            item.offsetWidth *
            .24
          );


        if(
          Math.abs(currentX) >= threshold
        ){

          dismissNotification(
            item,
            currentX < 0
            ?
            "left"
            :
            "right"
          );

        }else{

          item.classList.remove("ih-swiping");

          item.style.transform =
            "";

          item.style.opacity =
            "";

          updateNotificationStack();

        }

      }


      item.addEventListener(
        "pointerup",
        finishSwipe
      );


      item.addEventListener(
        "pointercancel",
        finishSwipe
      );

    }

    let activeDrag =
      null;


    folders.forEach(
      function(folder){

        let moved =
          false;

        let startX =
          0;

        let startY =
          0;

        let baseX =
          0;

        let baseY =
          0;


        folder.addEventListener(
          "pointerdown",
          function(event){

            if(
              event.button !== undefined &&
              event.button !== 0
            ){
              return;
            }


            moved =
              false;


            startX =
              event.clientX;

            startY =
              event.clientY;


            baseX =
              parseFloat(
                folder.dataset.dragX ||
                "0"
              );


            baseY =
              parseFloat(
                folder.dataset.dragY ||
                "0"
              );


            activeDrag =
              folder;


            folder.classList.add(
              "is-dragging"
            );


            try{

              folder.setPointerCapture(
                event.pointerId
              );

            }catch(error){}

          }
        );


        folder.addEventListener(
          "pointermove",
          function(event){

            if(
              activeDrag !== folder
            ){
              return;
            }


            const dx =
              event.clientX -
              startX;


            const dy =
              event.clientY -
              startY;


            if(
              Math.abs(dx) +
              Math.abs(dy) >
              7
            ){

              moved =
                true;

            }


            const x =
              baseX +
              dx;


            const y =
              baseY +
              dy;


            folder.dataset.dragX =
              x;


            folder.dataset.dragY =
              y;


            folder.style.translate =
              x +
              "px " +
              y +
              "px";


            if(trashDock){

              const folderRect =
                folder
                  .getBoundingClientRect();


              const trashRect =
                trashDock
                  .getBoundingClientRect();


              const overlap =
                !(
                  folderRect.right <
                    trashRect.left - 30 ||

                  folderRect.left >
                    trashRect.right + 30 ||

                  folderRect.bottom <
                    trashRect.top - 30 ||

                  folderRect.top >
                    trashRect.bottom + 30
                );


              trashDock.classList.toggle(
                "is-receiving",
                overlap
              );

            }


            updateHotspotPriority();

          }
        );


        folder.addEventListener(
          "pointerup",
          function(){

            if(
              activeDrag !== folder
            ){
              return;
            }


            folder.classList.remove(
              "is-dragging"
            );


            activeDrag =
              null;


            let dropped =
              false;


            if(
              trashDock &&
              trashDock.classList.contains(
                "is-receiving"
              )
            ){

              dropped =
                true;


              trashDock.classList.remove(
                "is-receiving"
              );


              folder.style.translate =
                "0px 0px";


              folder.dataset.dragX =
                "0";


              folder.dataset.dragY =
                "0";

                "trashModal"
              );

                "Trash",
                "Protected item. Confirmation required.",
                "⌫",
                3800
              );


              systemSound(
                "close"
              );

            }


            if(
              moved ||
              dropped
            ){

              folder.dataset.justDragged =
                "1";


              setTimeout(
                function(){

                  delete folder.dataset.justDragged;

                },
                700
              );

            }


            updateHotspotPriority();

          }
        );


        folder.addEventListener(
          "pointercancel",
          function(){

            folder.classList.remove(
              "is-dragging"
            );


            activeDrag =
              null;


            if(trashDock){

              trashDock.classList.remove(
                "is-receiving"
              );

            }


            updateHotspotPriority();

          }
        );

          "click",

        a.right <= b.left ||
        a.left >= b.right ||
        a.bottom <= b.top ||
        a.top >= b.bottom

    function waitForVideo(
      video
    ){

      return new Promise(
        function(resolve){

          let finished =
            false;


          function complete(ok){

            if(finished) return;


            finished =
              true;


            cleanup();

            resolve(ok);

          }


          function ready(){

            if(
              video.readyState >= 3
            ){

              complete(true);

            }

          }


          function fail(){

            complete(false);

          }


          function cleanup(){

            clearTimeout(
              timeout
            );


            video.removeEventListener(
              "canplay",
              ready
            );


            video.removeEventListener(
              "loadeddata",
              ready
            );


            video.removeEventListener(
              "error",
              fail
            );

          }


          const timeout =
            setTimeout(
              function(){

                complete(
                  video.readyState >= 2
                );

              },
              12000
            );


          video.addEventListener(
            "canplay",
            ready
          );


          video.addEventListener(
            "loadeddata",
            ready
          );


          video.addEventListener(
            "error",
            fail,
            {once:true}
          );


          ready();

        }
      );

    }


    function startPlayback(
      video
    ){

      return new Promise(
        function(resolve){

          let completed =
            false;


          function finish(ok){

            if(completed) return;


            completed =
              true;

            cleanup();

            resolve(ok);

          }


          function playing(){

            finish(true);

          }


          function timeUpdate(){

            if(
              video.currentTime > 0
            ){

              finish(true);

            }

          }


          function error(){

            finish(false);

          }


          function cleanup(){

            clearTimeout(
              timeout
            );


            video.removeEventListener(
              "playing",
              playing
            );


            video.removeEventListener(
              "timeupdate",
              timeUpdate
            );


            video.removeEventListener(
              "error",
              error
            );

          }


          const timeout =
            setTimeout(
              function(){

                finish(
                  !video.paused &&
                  video.readyState >= 2
                );

              },
              5000
            );


          video.addEventListener(
            "playing",
            playing,
            {once:true}
          );


          video.addEventListener(
            "timeupdate",
            timeUpdate
          );


          video.addEventListener(
            "error",
            error,
            {once:true}
          );


          try{

            video.currentTime =
              0;

          }catch(error){}


          video.muted =
            true;

          video.defaultMuted =
            true;


          try{

            const promise =
              video.play();


            if(
              promise &&
              promise.catch
            ){

              promise.catch(
                function(){

                  finish(false);

                }
              );

            }

          }catch(error){

            finish(false);

          }

        }
      );

    }

      );


      for(
        let index = 0;
        index < folders.length;
        index++
      ){

        folders[index].classList.add(
          "ivl-launched"
        );


        await wait(
          CONFIG.intro.folderDelay
        );

      }


      body.classList.add(
        "ivl-chrome"
      );


      await wait(
        CONFIG.intro.chromeDelay
      );


      body.classList.add(
        "ivl-dock"
      );


      await wait(
        CONFIG.intro.dockDelay
      );


      body.classList.add(
        "ivl-bar"
      );


      await wait(
        CONFIG.intro.finishDelay
      );


      body.classList.remove(
        "ivl-running",
        "ivl-chrome",
        "ivl-dock",
        "ivl-bar",
        "ih-monitor-start",
        "ih-monitor-bursting"
      );


      body.classList.add(
        "ivl-done",
        "ih-folders-launched",
        "ilnur-os-ready"
      );


      folders.forEach(
        function(folder){

          folder.classList.remove(
            "ivl-launched"
          );

        }
      );


      updateHotspotPriority();

    }


    function closeLoader(
      loader
    ){

      loader.classList.add(
        "is-hidden"
      );


      setTimeout(
        function(){

          if(loader.parentNode){

            loader.remove();

          }

        },
        560
      );

    }


    function freezeLastFrame(
      video
    ){

      video.pause();


      try{

        if(
          Number.isFinite(
            video.duration
          ) &&
          video.duration > 0
        ){

          video.currentTime =
            Math.max(
              0,
              video.duration - .04
            );

        }

      }catch(error){}


      video.classList.add(
        "is-visible"
      );

    }


    async function installIntro(){

      body.classList.add(
        "ivl-waiting"
      );


      body.classList.remove(
        "ivl-done"
      );


      const loader =
        createLoader();


      const video =
        createVideo();


      startProgress();


      let finished =
        false;


      const hardTimeout =
        setTimeout(
          function(){

            if(finished) return;


            finished =
              true;


            stopProgress();


            setProgress(
              100,
              "launching desktop..."
            );


            runIntro();


            closeLoader(
              loader
            );

          },
          16000
        );


      if(!video){

        clearTimeout(
          hardTimeout
        );


        finished =
          true;


        stopProgress();


        await runIntro();


        closeLoader(
          loader
        );


        startAfterIntro();


        return;

      }


      video.addEventListener(
        "ended",
        function(){

          freezeLastFrame(
            video
          );

        }
      );


      video.src =
        isMobile()
        ?
        CONFIG.mobileVideo
        :
        CONFIG.desktopVideo;


      video.load();


      const loaded =
        await waitForVideo(
          video
        );


      if(!loaded){

        clearTimeout(
          hardTimeout
        );


        if(!finished){

          finished =
            true;


          stopProgress();


          setProgress(
            100,
            "launching desktop..."
          );


          await runIntro();


          closeLoader(
            loader
          );


          startAfterIntro();

        }


        return;

      }


      setProgress(
        88,
        "starting video..."
      );


      const playing =
        await startPlayback(
          video
        );


      clearTimeout(
        hardTimeout
      );


      if(finished) return;


      finished =
        true;


      stopProgress();


      setProgress(
        100,
        "launching desktop..."
      );


      if(playing){

        video.classList.add(
          "is-visible"
        );

      }


      runIntro();


      await wait(90);


      closeLoader(
        loader
      );


      if(!playing){

        const retry =
          function(){

            video.muted =
              true;


            video.play()
              .then(function(){

                video.classList.add(
                  "is-visible"
                );

              })
              .catch(function(){});

          };


        document.addEventListener(
          "pointerdown",
          retry,
          {once:true}
        );


        document.addEventListener(
          "touchstart",
          retry,
          {
            once:true,
            passive:true
          }
        );

      }


      startAfterIntro();

    }


    /* =======================================================
       26. AFTER INTRO
    ======================================================= */

    let afterIntroStarted =
      false;


    function startAfterIntro(){

      if(afterIntroStarted){
        return;
      }


      afterIntroStarted =
        true;


      scheduleNotifications();

      updateHotspotPriority();

    }


    /* =======================================================
       27. INITIAL
    ======================================================= */

    folders.forEach(
      function(folder){

        folder.style.visibility =
          "visible";

      }
    );


    installIntro();

  }

})();
</script>
